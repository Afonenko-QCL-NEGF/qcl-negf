function assert(value: unknown, message = "Assertion failed"): asserts value {

function equal(actual: unknown, expected: unknown): void {

function rejects(action: () => unknown): void {
  let rejected = false;
  try {
    action();
  } catch {
    rejected = true;
  }
  assert(rejected, "Invalid input was accepted");
}

// Test-only relocation of otherwise fixed facts. No production path algorithm,
// hashing or ownership inference lives here. The newly manufactured admission
// digest is rebound on all three representations of that retained reference.
function relocatedStorage(
  location: keyof ReturnType<typeof descriptor>["storage"],
  value: string,
) {
  const d = descriptor();
  const m = metadata();
  const previous = d.storage[location];
  const relocated = (path: string) => path === previous ? value :
    path.startsWith(previous + "/") ? value + path.slice(previous.length) : path;
  d.storage[location] = value;
  m.directories = Object.fromEntries(Object.entries(m.directories).map(([path, fact]) => [
    relocated(path), { ...fact, canonical: relocated(fact.canonical) },
  ]));
  m.absent_paths = m.absent_paths.map(relocated);
  m.admission.exclusive_storage = m.admission.exclusive_storage.map(relocated);
  d.admission_ref.sha256 = OTHER;
  m.files[d.admission_ref.path].sha256 = OTHER;
  m.admission.ref.sha256 = OTHER;
  return { d, m };
}

function nodeTap(count: number): string {
  return "TAP version 13\n" + Array.from({ length: count }, (_, i) =>
    `# Subtest: case ${i + 1}\nok ${i + 1} - case ${i + 1}\n`).join("") +
    `1..${count}\n# tests ${count}\n# suites 0\n# pass ${count}\n` +
    "# fail 0\n# cancelled 0\n# skipped 0\n# todo 0\n# duration_ms 1\n";
}
function denoTap(count: number): string {

}

  return {
    stdout: name === "doc04-node" ? nodeTap(15) : name === "doc06-deno" ? denoTap(7) :

    exit_code: 0, signal: null,
    started_at: "2026-10-20T00:00:00.000Z", ended_at: "2026-10-20T00:00:01.000Z",
    capture_complete: true, drains_complete: true, stop_reason: null,
    supervision: {
      process_tree_owned: true, limits_enforced: true, evidence_ref: ref("supervision"),
    },
  };
}
type Step = ReturnType<typeof planChecks>["steps"][number];
function fakeExecutor(change?: (step: Step, capture: Capture) => Capture) {
  const calls: Step[] = [];
  let graph = HASH;
  return {
    calls,
    setGraph(value: string) { graph = value; },
    sourceGraph() { return { revision: REVISION, digest: graph, clean: true }; },
    async execute(step: Step) {
      calls.push(structuredClone(step));
      const capture = healthyCapture(step.name);
      return change ? change(step, capture) : capture;
    },
  };
}

function unitPass(format: string, stdout: string, stderr: string, expected: number): boolean {
  try {
    return parseUnitReport(format, stdout, stderr, expected).result === "passed_unit";
  } catch {
    return false;
  }
}

    const result = preflightInputs(parseInputs(JSON.stringify(d)), m);
    equal(result.result, "ready_for_unit_attempt");

      equal(m.directories[d.storage[location]].canonical, d.storage[location]);

        `${d.storage.owned_tmp_root}/node`,
